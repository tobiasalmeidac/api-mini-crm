# ------------------------------------------------------------
# 1) TOPO DO ARQUIVO: importar o que falta e guardar o token
# ------------------------------------------------------------

importar Header, Depends (do fastapi)

TOKEN = "meu-token"      # token esperado (em projeto real vai em variável de ambiente)


# ------------------------------------------------------------
# 2) A FUNÇÃO QUE VERIFICA (escrita UMA vez só)
#    Ela segue exatamente o fluxograma:
#    tem Authorization? -> tem Bearer? -> token válido?
# ------------------------------------------------------------

def verificar_token(authorization = Header(padrão=None)):

    # o FastAPI pega o header "Authorization" e entrega aqui.
    # Se o cliente não mandou nada, authorization vale None.

    # PERGUNTA 1: tem Authorization?
    se authorization for None:
        levantar erro 401 com mensagem "Authorization ausente"

    # PERGUNTA 2: tem Bearer?
    # o valor chega como texto único, por exemplo: "Bearer meu-token"
    se authorization NÃO começar com "Bearer ":
        levantar erro 401 com mensagem "Formato inválido, use Bearer"

    # separar o token do resto do texto
    # "Bearer meu-token"  ->  "meu-token"
    token_recebido = authorization sem os 7 primeiros caracteres ("Bearer ")

    # PERGUNTA 3: token é válido?
    se token_recebido for diferente de TOKEN:
        levantar erro 401 com mensagem "Token inválido"

    # se chegou até aqui, passou nas 3 perguntas.
    # não retorna nada e não levanta erro: o FastAPI deixa a rota rodar.


# ------------------------------------------------------------
# 3) USAR NAS ROTAS
# ------------------------------------------------------------

#         +-- token certo ........... segue para listar_clientes -> 200
